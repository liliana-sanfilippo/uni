\section{Genetische Kartierung}
\subsection{Einführung}

\begin{itemize}
    \item \textbf{Genkarte:} lineare Anordnung der Gene im Genom
    \item \textbf{Genetische Genkarte:} durch genetische Kartierung erstellt
    \item \textbf{Physikalische Genkarte:} molekulare Karte, durch Sequenzierung erstellt
\end{itemize}

\begin{center}
    genetische Karte $\neq$ physikalische Karte!
\end{center}
