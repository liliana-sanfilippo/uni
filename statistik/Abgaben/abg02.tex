(b) Seien $(\Omega,P)$ ein Wahrscheinlichkeitsraum und $A,B\subseteq\Omega$ mit $P(A)=P(B)=\frac45$.
\begin{auf}[(i) Finden Sie eine Zahl $c\in(0,1)$ derart, dass $P(A\cap B)\ge c$.]
    $\ $
    \begin{align*}
        \pe(A \backslash B) & \leq c \\
        \pe(A \backslash B) &  = \pe(A) - \pe(A \cap B) \\
        \pe(A \cap B) & \geq \pe(A) + \pe(B) -1 \\
        \Leftrightarrow \pe(A \cap B) & \geq \frac45 + \frac45 -1  = \frac35
    \end{align*}
    \[
        \pe(A \backslash B)  = \pe(A) - \pe(A \cap B) \leq \frac45 + \frac35  = \frac15 \quad \Rightarrow c = \frac15

\end{auf}
