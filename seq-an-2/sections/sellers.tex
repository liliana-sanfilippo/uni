\subsection*{proceed column-by-column}

\begin{multicols}{2}
    \noindent \textbf{\underline{Der Pseudocode:}}
    \begin{algorithmic}
        \For{$j \gets 1,...,n$}
            \State $C_j(0) \gets 0$
            \For{$i \gets 1,...,m$}
                \State $C_j (i) \gets min \begin{cases}
                                              C_{j-1}(i-1) + cost(x[i], y[j]) \\
                                              C_{j-1}(i) + \gamma \\ C_j (i-1) + \gamma
                \end{cases}$
            \EndFor
            \If{$C_j(m) \leq k$}
                \Statex \textbf{report} $(j, C_j (m))$
            \EndIf
        \EndFor
    \end{algorithmic}
    \columnbreak
    \textbf{\underline{Wird mit Einsetzen zu:}}
    \begin{algorithmic}
        \For{$j \gets 1,...,5$}
            \State $C_j(0) \gets 0$
            \For{$i \gets 1,...,3$}
                \State $C_j (i) \gets min \begin{cases}
                                              C_{j-1}(i-1) + cost(x[i], y[j]) \\
                                              C_{j-1}(i) + 1 \\ C_j (i-1) + 1
                \end{cases}$
            \EndFor
            \If{$C_j(3) \leq 2$}
                \Statex \textbf{report} $(j, C_j (3))$
            \EndIf
        \EndFor
    \end{algorithmic}
\end{multicols}
