\newglossaryentry{rising_unranking}{
    name={Rising unranking function},
    description={rising_unranking},
    formB={\frac{r}{\sigma} = r', \quad  \text{R } c}
}

\newglossaryentry{falling_unranking}{
    name={Falling unranking function},
    description={falling_unranking},
    formB={\frac{r}{\sigma^{q-1}} = c, \quad  \text{R } r'}
}

\newglossaryentry{falling_ranking}{
    name={falling ranking function},
    description={falling_ranking},
    formB={r(z) =
    \Sigma^q_{i=1}
    r_{\Sigma}
    (z[i]) \cdot \sigma^{q-i}}
}

\newglossaryentry{rising_ranking}{
    name={rising ranking function},
    description={rising_ranking},
    formB={r(z) =
    \Sigma^q_{i=1}
    r_{\Sigma}
    (z[i]) \cdot \Sigma^{i-1}}
}

\newglossaryentry{qgdis_formel}{
    name={q-Gram Distanz},
    description={qgdis_formel},
    formB={d_q(x, y) := \sum_{z\in\Sigma^q}
    |p_q(x)_z - p_q(y)_z|}
}
