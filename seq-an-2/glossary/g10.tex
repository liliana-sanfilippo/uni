\newglossaryentry{optimal_cut}{
    name={optimal cut},
    description={A family of cut positions $(c_1, c_2, \ldots, c_k)$ of the sequences $s_1, s_2, \ldots, s_k$ of lengths
    $n_1, n_2, \ldots, n_k$ for which exists an
    alignment $A$ of the prefixes $(s_1[1\ldots c_1], s_2[1\ldots c_2], \ldots, s_k[1\ldots c_k])$ and an
    alignment $B$ of the suffixes $(s_1[c_1+1\ldots n_1], s_2[c_2+1\ldots n_2], \ldots, s_k[c_k+1\ldots
    n_k])$ such that the concatenation $A++B$ is an optimal alignment of $s_1, s_2, \ldots,
    s_k$.},
    type=defi
}
