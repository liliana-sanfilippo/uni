\newglossaryentry{qgprof}{
name={q-gram profile},
description={Let $\sigma = |\sum|$. The q-gram profile of x is a $\sigma^q$-element vector $p_q(x) := (p_q(x)_z)_{z\in\sum^q}$, indexed by all q-grams, defined by $p_q(x)_z := Occ(x, z)$},
type=defi,
nominative={$O(\#q-grams + m + n)$ or $O(n)$ if m $\leq$ n and $q = \Bigg\lfloor \frac{\log(m) − \log(
c)}{\log(\sigma)} \Bigg\rfloor$},
accusative={Aufgabe},
genitive={$O(\#q-grams)$}
}

description={The q-gram distance of $x \in \Sigma^m$ and $y \in \Sigma^n$ is defined for $1 \leq q \leq min\{m, n\}$
as: $d_q(x, y) := \sum_{z\in\Sigma^q} |p_q(x)_z - p_q(y)_z |$},
type=defi,
symbol={\ensuremath{d_q}},
altsymbol={d},
formF={The q-gram distance of $x \in \Sigma^m$ and $y \in \Sigma^n$ is defined for $1 \leq q \leq min\{m, n\}$
as:},
formB={

|p_q(x)_z - p_q(y)_z |

}
